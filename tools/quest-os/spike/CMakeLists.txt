cmake_minimum_required(VERSION 3.22)
project(anyps5_spike C CXX)
set(OPENXR_SDK "" CACHE PATH "OpenXR-SDK source")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
add_subdirectory(${OPENXR_SDK} openxr EXCLUDE_FROM_ALL)
add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(anyps5spike SHARED main.c)
target_link_libraries(anyps5spike native_app_glue openxr_loader vulkan android log)
set_target_properties(anyps5spike PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate")
